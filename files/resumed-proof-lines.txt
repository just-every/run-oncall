     1	import assert from 'node:assert/strict';
     2	import {randomUUID} from 'node:crypto';
     3	import {agentEffectSnapshot} from './agent-door-runtime.mjs';
     4	import {autonomousPolicy,heldPolicy} from '../src/modules/agent-door/conformance/crypto.mjs';
     5	
     6	/** A completed Scribe assignment rejects a changed reason and retains its first outcome. */
     7	export async function changedScribeReasonConflict(runtime){
     8	 const {db,http,tool,adopt,context}=runtime;
     9	 const read=async path=>{const response=await http(path);assert.equal(response.status,200);return response.json()};
    10	 const command=async input=>{const response=await http('/api/command',input);assert.equal(response.status,200);return response.json()};
    11	 const rows=async(sql,...values)=>(await db.prepare(sql).bind(...values).all()).results;
    12	 const snapshot=()=>agentEffectSnapshot(runtime);
    13	 const audits=()=>rows('SELECT * FROM agent_door_audit ORDER BY sequence');
    14	 const permanent=()=>rows('SELECT * FROM oc_admin_audit ORDER BY ordinal');
    15	 const owner=(await read('/api/workspace')).member;
    16	 const name='oncall.incidents.update';
    17	 for(const [mode,policy,disposition] of [['autonomous',autonomousPolicy(),'autonomous'],['held',heldPolicy(),'review']]){
    18	  assert.equal((await adopt(policy)).status,200);
    19	  const incident=await command({type:'declare',service:'agent-service',title:'Synthetic Scribe reason conflict '+mode,
    20	   summary:'Native coordination without protected evidence',severity:'warning',alerts:[]});
    21	  const before=await snapshot(),pagingBefore=await read('/api/paging/state'),priorAudit=await audits();
    22	  assert.equal(incident.scribe??null,null,'A fresh incident has no Scribe');
    23	  assert.equal(incident.state,'declared');
    24	  const responder=before.application.members.find(row=>row.active&&row.role==='responder');
    25	  assert.ok(responder,'Scribe assignment must name an existing active responder');
    26	  assert.deepEqual(incident.alerts,[]);assert.deepEqual(incident.actions,[]);assert.equal(incident.recovery,'');
    27	  const args=Object.freeze({change:'assign-role',role:'scribe',incident:incident.id,revision:incident.revision,member:responder.id,
    28	   reason:'Synthetic literal Scribe reason <keep this text> & retain it',idempotency_key:randomUUID()});
    29	  const first=await tool(name,args);
    30	  assert.equal(first.status,200);assert.equal(first.body.error,undefined);assert.equal(first.result.isError,false);
    31	  assert.deepEqual(first.decision.classes.map(row=>row.class),['standard']);assert.equal(first.decision.disposition,disposition);
    32	  const once=await snapshot(),result=first.result.structuredContent,pagingOnce=await read('/api/paging/state');
    33	  const native=JSON.parse(once.records.oc_incidents.find(row=>row.id===incident.id).data);
    34	  assert.deepEqual(result,native);assert.deepEqual(once.application.incidents.find(row=>row.id===incident.id),native);
    35	  assert.equal(result.scribe,responder.id);
    36	  assert.deepEqual(result,{...incident,scribe:responder.id,revision:incident.revision+1,timeline:result.timeline},'Only Scribe, revision and the appended timeline entry change');
    37	  assert.equal(result.revision,args.revision+1);assert.equal(once.application.revision,before.application.revision+1);
    38	  assert.equal(once.records.workspace_control[0].revision,before.records.workspace_control[0].revision+1);
    39	  assert.equal(once.application.incidents.length,before.application.incidents.length);
    40	  assert.deepEqual(once.application.incidents.filter(row=>row.id!==incident.id),before.application.incidents.filter(row=>row.id!==incident.id));
    41	  assert.deepEqual(result.timeline.slice(0,-1),incident.timeline);assert.equal(result.timeline.length,incident.timeline.length+1);
    42	  const entry=result.timeline.at(-1);
    43	  assert.equal(entry.kind,'assign-role');assert.equal(entry.text,'scribe: Unassigned → '+responder.name+': '+args.reason);
    44	  assert.equal(entry.actor,context.agentId);assert.notEqual(entry.actor,responder.id);
    45	  assert.equal(entry.metadata.agent.agent.id,context.agentId);assert.equal(entry.metadata.agent.for.id,owner.id);
    46	  assert.equal(result.timeline.filter(row=>row.kind==='assign-role').length,1);
    47	  assert.deepEqual(result.actions,incident.actions);assert.deepEqual(result.alerts,incident.alerts);
    48	  const commits=once.records.oc_admin_audit.filter(row=>!before.records.oc_admin_audit.some(prior=>prior.id===row.id));
    49	  assert.equal(commits.length,1,'Exactly one native state-changing audit commits the Scribe assignment');
    50	  const commit=commits[0];assert.equal(commit.kind,'incidents.update');assert.equal(commit.outcome,'accepted');
    51	  assert.equal(commit.workspace_before,before.application.revision);assert.equal(commit.workspace_after,once.application.revision);
    52	  assert.equal(commit.paging_before,commit.paging_after);assert.equal(commit.handoff_before,commit.handoff_after);
    53	  for(const [field,kind,id] of [['actor','agent',context.agentId],['subject','entity',incident.id],['source','member',owner.id]]){
    54	   const identity=await db.prepare('SELECT label FROM oc_audit_identities WHERE kind=? AND entity_id=?').bind(kind,id).first();
    55	   assert.equal(commit[field],identity.label,'The native commit retains its '+field+' attribution');
    56	  }
    57	  for(const prior of before.records.oc_admin_audit)assert.deepEqual(once.records.oc_admin_audit.find(row=>row.id===prior.id),prior);
    58	  assert.deepEqual(once.records.agent_door_approvals,before.records.agent_door_approvals);
    59	  assert.equal(once.records.agent_door_calls.length,before.records.agent_door_calls.length+1);
    60	  for(const prior of before.records.agent_door_calls)assert.deepEqual(once.records.agent_door_calls.find(row=>row.agent===prior.agent&&row.key===prior.key),prior);
    61	  const calls=await rows('SELECT * FROM agent_door_calls WHERE agent=? AND key=?',context.agentId,args.idempotency_key);
    62	  assert.equal(calls.length,1);assert.equal(calls[0].state,'done');assert.equal(calls[0].decision_id,first.decision.id);
    63	  const stored=JSON.parse(calls[0].outcome);
    64	  assert.equal(stored.outcome,'done');assert.deepEqual(stored.result,result);
    65	  assert.deepEqual(stored.review,first.result._meta['com.runeditrun/review']);
    66	  const reviews=once.records.agent_door_reviews;
    67	  assert.equal(reviews.length,before.records.agent_door_reviews.length+Number(mode==='held'));
    68	  for(const prior of before.records.agent_door_reviews)assert.deepEqual(reviews.find(row=>row.id===prior.id),prior);
    69	  if(mode==='held'){
    70	   const review=reviews.find(row=>row.id===stored.review.id);
    71	   assert.ok(review);assert.equal(review.state,'open');assert.equal(JSON.parse(review.saved).decision_id,first.decision.id);
    72	   assert.equal((await read('/agents/api/reviews')).reviews.find(row=>row.id===review.id).state,'open');
    73	  }else assert.equal(stored.review,undefined);
    74	  for(const table of Object.keys(before.records).filter(table=>table.startsWith('paging_')))assert.deepEqual(once.records[table],before.records[table]);
    75	  assert.deepEqual(pagingOnce,{...pagingBefore,workspaceRevision:once.application.revision});
    76	  const originalAudit=await audits();assert.deepEqual(originalAudit.slice(0,-1),priorAudit);assert.equal(originalAudit.length,priorAudit.length+1);
    77	  const original=JSON.parse(originalAudit.at(-1).row);
    78	  assert.equal(original.id,first.decision.id);assert.equal(original.tool,name);assert.equal(original.outcome,'done');
    79	  assert.equal(original.idempotency_key,args.idempotency_key);assert.equal(original.replay_of,null);
    80	  assert.equal(original.agent.id,context.agentId);assert.equal(original.sponsor.id,owner.id);
    81	  const permanentOnce=await permanent(),relayOnce=await rows('SELECT * FROM agent_door_relay ORDER BY sequence');
    82	  // Both calls below retain the original key and now-stale native revision.
    83	  const changed={...args,reason:'Another schema-valid literal Scribe reason'};
    84	  assert.deepEqual({...changed,reason:args.reason},args,'Only the reason changes');
    85	  const conflict=await tool(name,changed);
    86	  assert.equal(conflict.status,200);assert.equal(conflict.body.error,undefined);assert.notEqual(conflict.body.id,first.body.id);
    87	  assert.equal(conflict.result.isError,true);assert.equal(conflict.result._meta['com.runeditrun/error'].code,'idempotency_conflict');
    88	  assert.equal(JSON.parse(conflict.result.content[0].text).code,'idempotency_conflict');assert.equal(conflict.result.structuredContent,undefined);
    89	  assert.deepEqual(await snapshot(),once,'Changed reason preserves native records, revisions, timeline, permanent state-changing audit, original calls and outcomes, approvals, reviews and paging');
    90	  assert.deepEqual(await read('/api/paging/state'),pagingOnce);
    91	  const refused=await audits();assert.deepEqual(refused.slice(0,-1),originalAudit);assert.equal(refused.length,originalAudit.length+1);
    92	  const refusal=JSON.parse(refused.at(-1).row);
    93	  assert.equal(refusal.tool,name);assert.equal(refusal.outcome,'refused');assert.equal(refusal.code,'idempotency_conflict');
    94	  assert.notEqual(refusal.id,first.decision.id);
    95	  assert.equal(refusal.idempotency_key,args.idempotency_key);assert.equal(refusal.agent.id,context.agentId);assert.equal(refusal.sponsor.id,owner.id);
    96	  assert.equal(refusal.review,null);assert.equal(refusal.approval,null);
    97	  assert.deepEqual(await rows('SELECT * FROM agent_door_calls WHERE agent=? AND key=?',context.agentId,args.idempotency_key),calls);
    98	  const replay=await tool(name,args);
    99	  assert.equal(replay.status,200);assert.equal(replay.body.error,undefined);assert.equal(replay.result.isError,false);
   100	  assert.notEqual(replay.body.id,first.body.id);assert.notEqual(replay.body.id,conflict.body.id);
   101	  assert.deepEqual(replay.result.structuredContent,stored.result);assert.deepEqual(replay.result._meta['com.runeditrun/review'],stored.review);
   102	  assert.deepEqual(await snapshot(),once,'Exact original arguments return the saved outcome with no second Scribe assignment, revision, state-changing audit or held review');
   103	  assert.deepEqual(await read('/api/paging/state'),pagingOnce);
   104	  assert.deepEqual(await rows('SELECT * FROM agent_door_calls WHERE agent=? AND key=?',context.agentId,args.idempotency_key),calls);
   105	  const retried=await audits();assert.deepEqual(retried.slice(0,-1),refused);assert.equal(retried.length,refused.length+1);
   106	  const retry=JSON.parse(retried.at(-1).row);
   107	  assert.equal(retry.id,replay.decision.id);assert.equal(retry.tool,name);assert.equal(retry.outcome,'replayed');
   108	  assert.equal(retry.replay_of,first.decision.id);assert.equal(retry.operation_id,original.operation_id);
   109	  assert.equal(retry.idempotency_key,args.idempotency_key);assert.equal(retry.agent.id,context.agentId);assert.equal(retry.sponsor.id,owner.id);
   110	  const relayAfter=await rows('SELECT * FROM agent_door_relay ORDER BY sequence');
   111	  assert.deepEqual(relayAfter.slice(0,relayOnce.length),relayOnce);
   112	  assert.deepEqual(relayAfter.slice(relayOnce.length).map(row=>row.decision_id),[refusal.id,retry.id]);
   113	  if(mode==='held'){
   114	   const review=(await read('/agents/api/reviews')).reviews.find(row=>row.id===stored.review.id);
   115	   assert.equal(review.state,'open');assert.equal(review.decision_id,first.decision.id);
   116	  }
   117	  const permanentAfter=await permanent();assert.deepEqual(permanentAfter.slice(0,permanentOnce.length),permanentOnce);
   118	  const bookkeeping=permanentAfter.slice(permanentOnce.length);assert.ok(bookkeeping.length>0);
   119	  for(const record of bookkeeping){
   120	   assert.equal(record.kind,'agent-door.persistence');assert.ok(['accepted','refused'].includes(record.outcome));
   121	   for(const family of ['workspace','paging','handoff'])assert.equal(record[family+'_before'],record[family+'_after'],'Attempt audit never commits another application effect');
   122	  }
   123	  assert.deepEqual(await runtime.outboundRequests(),[],'Scribe assignment, changed reason and exact retry attempt zero outbound requests');
   124	  console.log('PROBE BASE-AGENT-012 acceptance 3 '+mode+' Scribe assignment: one standard-only agent-attributed assign-role from unassigned to an existing active responder, one incident and workspace revision and native audit commit; original key and revision with changed reason -> isError idempotency_conflict in metadata and text, no private result; exact original arguments with new RPC id -> saved result and original review; native effect custody and paging unchanged; attributed refusal and linked replay audit only; zero outbound requests');
   125	 }
   126	 assert.equal((await adopt()).status,200);
   127	}
